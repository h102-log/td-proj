extends SceneTree
# 일회성 캡처(조작감 FB 갈래). 맵 뷰(battle_view)만 띄워 장면마다 SubViewport 그림을 PNG로 저장한다. main·HUD·설정은 띄우지 않는다.
# 사용: godot --path <snap> -s <이 파일> -- <출력 폴더>

const Main := preload("res://src/main.gd")
const Battle := preload("res://src/sim/battle.gd")
const Controls := preload("res://src/ui/controls.gd")
const View := preload("res://src/ui/battle_view.gd")
const BIG := 1000000
const SCENES := 5

var out: String
var sv: SubViewport
var view
var b
var c
var k := 0
var wait := 0

func _initialize() -> void:
	out = OS.get_cmdline_user_args()[0].trim_suffix("/") + "/"
	var box := Main.map_viewport(root)
	root.add_child(box)
	sv = box.get_child(0)
	b = Battle.new({"size": Vector2i(30, 20), "base": Vector2i(25, 10), "base_hp": 100, "spawns": [Vector2i(0, 10)], "gold": BIG,
		"waves": [[{"kind": "basic", "count": 0}]], "heroes": [{"kind": "aias", "cell": Vector2i(8, 8)}, {"kind": "vesper", "cell": Vector2i(12, 8)}]})
	b.spawn_enemy("basic", Vector2i(16, 8), {"speed": 0, "damage": 0, "hp": BIG})
	b.start_battle()
	view = View.new()
	view.battle = b
	c = Controls.new(b, View.CELL)
	view.controls = c
	sv.add_child(view)
	var cam := Camera2D.new()
	cam.zoom = Vector2.ONE * 2
	cam.position = Vector2(12.5, 10.0) * View.CELL
	sv.add_child(cam)

func _scene(i: int) -> void:
	var aias = b.heroes[0]
	var vesper = b.heroes[1]
	match i:
		1:  # 고른 아이아스 · 안 고른 베스퍼(보호막)
			c.selected = [aias]
			b._shield(vesper, "test", 50.0, 99.0, 50.0)
		2:  # 우클릭 이동 0.1초 뒤
			b.order_move([aias], Vector2i(8, 12))
			b.step(0.05)
			b.step(0.05)
		3:  # 베스퍼 E(적 대상) 조준, 커서 빈 땅
			c.selected = [vesper]
			c.aiming = "e"
			c.aim_hero = vesper
			c.mouse = Vector2(16.5, 11.5) * View.CELL
		4:  # 커서 적 위
			c.mouse = Vector2(16.5, 8.5) * View.CELL
		5:  # 커서 사거리(6칸) 밖 빈 땅
			c.mouse = Vector2(19.5, 13.5) * View.CELL

func _process(_d: float) -> bool:
	if wait > 0:
		wait -= 1
		return false
	if k > 0:
		sv.get_texture().get_image().save_png(out + "fb_%d.png" % k)
	if k >= SCENES:
		quit(0)
		return true
	k += 1
	_scene(k)
	view.queue_redraw()
	wait = 3
	return false
