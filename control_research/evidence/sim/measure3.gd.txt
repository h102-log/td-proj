extends SceneTree
# 일회성 측정 3: 유닛 중간 waypoint 도착 규칙을 적과 같게(칸에 들어서면 도착) 바꾼 사본과 원본 비교
# 인자: orig | mod

const DT := 1.0 / 60.0
const BIG := 1000000
var SRC := "res://src/sim/battle.gd"

func B(over := {}):
	var stage := {"size": Vector2i(30, 20), "base": Vector2i(25, 10), "base_hp": 100,
		"spawns": [Vector2i(0, 10)], "gold": BIG, "waves": [], "hero_start": null}
	stage.merge(over, true)
	return load(SRC).new(stage)

func P(s: String) -> void:
	print("RESULT ", s)

func run(b, secs: float) -> void:
	var t := 0.0
	while t < secs:
		b.step(DT)
		t += DT

func _initialize() -> void:
	var which: String = OS.get_cmdline_user_args()[0]
	if which == "mod":
		SRC = OS.get_cmdline_user_args()[1]
	P("variant=" + which)
	var line := []
	for off in [0.0, 0.2, 0.35, 0.5, 0.55]:
		var b = B({"hero_start": Vector2i(5, 5)})
		var h = b.hero
		var e = b.spawn_enemy("basic", Vector2i(8, 5), {"speed": 0, "sight": 0, "damage": 0, "hp": BIG})
		e.pos = Vector2(8, 5 + off)
		b.order_move([h], Vector2i(12, 5))
		var t := 0.0
		while t < 5.0 and h.pos.distance_to(Vector2(12, 5)) > 0.01:
			b.step(DT)
			t += DT
		line.append("off=%.2f:%s@%.2fs(%.2f,%.2f)hit=%s" % [off, "reach" if h.pos.distance_to(Vector2(12, 5)) <= 0.01 else "STOP", t, h.pos.x, h.pos.y, e.hp < e.max_hp])
	P("offset: " + " ".join(PackedStringArray(line)))
	# 대각선
	var b = B({"hero_start": Vector2i(5, 5)})
	var h = b.hero
	b.spawn_enemy("basic", Vector2i(8, 8), {"speed": 0, "sight": 0, "damage": 0, "hp": BIG})
	b.order_move([h], Vector2i(11, 11))
	run(b, 5.0)
	P("diagonal: end=(%.2f,%.2f) reached=%s" % [h.pos.x, h.pos.y, h.pos.distance_to(Vector2(11, 11)) < 0.05])
	# 세 적 줄
	var b3 = B({"hero_start": Vector2i(5, 5)})
	var h3 = b3.hero
	for y in [4, 5, 6]:
		b3.spawn_enemy("basic", Vector2i(8, y), {"speed": 0, "sight": 0, "damage": 0, "hp": BIG})
	b3.order_move([h3], Vector2i(12, 5))
	run(b3, 5.0)
	P("3-line: end=(%.2f,%.2f) reached=%s path=%d" % [h3.pos.x, h3.pos.y, h3.pos.distance_to(Vector2(12, 5)) < 0.05, h3.path.size()])
	# EV385꼴: 적이 선 칸으로
	var b4 = B({"hero_start": Vector2i(5, 5)})
	var h4 = b4.hero
	var e4 = b4.spawn_enemy("basic", Vector2i(9, 5), {"speed": 0, "sight": 0, "damage": 0, "hp": BIG})
	b4.order_move([h4], Vector2i(9, 5))
	run(b4, 3.0)
	P("onto enemy cell: path=%d hit=%s end=(%.2f,%.2f)" % [h4.path.size(), e4.hp < e4.max_hp, h4.pos.x, h4.pos.y])
	# 이그니스 지점 스킬 10칸, 길에 적(옆으로 0.3 비킴)
	var wave := [[{"kind": "basic", "count": 0}]]
	var b5 = B({"heroes": [{"kind": "ignis", "cell": Vector2i(5, 5)}], "waves": wave})
	var h5 = b5.hero
	var e5 = b5.spawn_enemy("basic", Vector2i(7, 5), {"speed": 0, "sight": 0, "damage": 0, "hp": BIG})
	e5.pos = Vector2(7, 5.3)
	b5.start_battle()
	b5.cast(h5, "q", Vector2(15, 5))
	run(b5, 4.0)
	P("ignis Q 10 cells, enemy offset 0.3 at 2: uses=%d end=(%.2f,%.2f)" % [h5.skill_uses, h5.pos.x, h5.pos.y])
	# 실제 스테이지 이동 시간(회귀 확인)
	var st: Dictionary = load("res://src/main.gd").stage()
	st["heroes"] = [{"kind": "aias", "cell": Vector2i(19, 11)}]
	for target in [Vector2i(1, 30), Vector2i(35, 20)]:
		var b6 = load(SRC).new(st)
		var h6 = b6.hero
		b6.order_move([h6], target)
		var t6 := 0.0
		while t6 < 30.0 and h6.pos.distance_to(Vector2(target)) > 0.01:
			b6.step(DT)
			t6 += DT
		P("stage walk to %s: %.2fs" % [target, t6])
	var b7 = load(SRC).new(st)
	b7.set_gate(b7.castle[Vector2i(20, 9)], true)
	var h7 = b7.hero
	b7.order_move([h7], Vector2i(10, 9))
	var t7 := 0.0
	while t7 < 30.0 and h7.pos.distance_to(Vector2(10, 9)) > 0.01:
		b7.step(DT)
		t7 += DT
	P("stage walk to wall top (10,9) via open gate: %.2fs on_wall=%s" % [t7, h7.on_wall])
	P("DONE")
	quit(0)
